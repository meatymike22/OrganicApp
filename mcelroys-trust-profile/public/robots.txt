# Rootify crawler rules.

User-agent: *

# Filtered search is an unbounded URL space and is not useful to index.
# Bare /search stays crawlable so the page itself can rank.
Disallow: /search?
Disallow: /*?page=
Disallow: /*?q=

Disallow: /api/

# Honoured by Bing, Yandex and most smaller crawlers. Google ignores it;
# rate-limit Googlebot in Search Console instead.
Crawl-delay: 10
